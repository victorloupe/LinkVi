Configuração Inicial do Banco

⚠️ Tabela Não Encontrada: Certifique-se de criar a nova tabela stores no Supabase SQL Editor. 
create table if not exists stores (
  id text primary key,
  password text not null,
  config jsonb not null
);

-- Inserir o Admin Inicial
insert into stores (id, password, config)
values ('contato@linksvi.com.br', 'admin123', '{}'::jsonb)
on conflict (id) do nothing;

-- RLS: leitura pública de id/config (necessário para as páginas de link),
-- mas a coluna password NUNCA é lida diretamente — veja supabase_security_hardening.sql
-- na raiz do projeto para as funções de login_store / admin_reveal_password
-- e para as políticas completas (incluindo escrita).
alter table stores enable row level security;
create policy "Allow public read access" on stores for select using (true);
create policy "Allow public insert access" on stores for insert with check (true);
create policy "Allow public update access" on stores for update using (true);
create policy "Allow public delete access" on stores for delete using (true);
revoke select on stores from anon, authenticated;
grant select (id, config) on stores to anon, authenticated;
-- Depois de rodar isso, rode também o conteúdo de supabase_security_hardening.sql.

📦 SUPABASE STORAGE: Crie um bucket público chamado logos (Storage -> New Bucket -> Nome: "logos" -> Marque como "Public").

Para liberar os uploads de imagem neste bucket, copie e execute este código no SQL Editor do Supabase: 

-- Liberar uploads na pasta logos
CREATE POLICY "Permitir upload público" ON storage.objects FOR INSERT WITH CHECK (bucket_id = 'logos');
CREATE POLICY "Permitir update público" ON storage.objects FOR UPDATE USING (bucket_id = 'logos');
